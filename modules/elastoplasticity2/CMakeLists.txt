# ----------------------------------------------------------------------------
# Support of the Von Mises law through the MGIS (MFrontGenericInterfaceSupport)
# library
# ----------------------------------------------------------------------------
set(ARCANEFEM_HAS_MGIS OFF)
find_package(MFrontGenericInterface QUIET)
find_program(MFRONT_EXECUTABLE mfront)

if (MFrontGenericInterface_FOUND AND MFRONT_EXECUTABLE)
  set(ARCANEFEM_HAS_MGIS ON)
  message(STATUS "Elastoplasticity2: MGIS support enabled"
         " (mgis: ${MFrontGenericInterface_DIR}, mfront: ${MFRONT_EXECUTABLE})")
  target_link_libraries(Elastoplasticity2 PRIVATE mgis::MFrontGenericInterface)
  target_compile_definitions(Elastoplasticity2 PRIVATE ARCANEFEM_HAS_MGIS)
  set_target_properties(Elastoplasticity2 PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF)

  # --------------------------------------------------------------------------
  # Generation and compilation of the Von Mises MFront behaviour library.
  # The sources are generated by mfront in the build directory and compiled
  # as a shared library next to the Elastoplasticity2 executable, so that
  # the default 'libVonMises.so' option of the arc files is resolved.
  # --------------------------------------------------------------------------
  # mfront is executed in an environment ensuring that the TFEL libraries
  # and the libstdc++ of the compiler used to build them are found.
  get_filename_component(MFRONT_BIN_DIR ${MFRONT_EXECUTABLE} DIRECTORY)
  get_filename_component(MFRONT_INSTALL_DIR ${MFRONT_BIN_DIR} DIRECTORY)
  get_filename_component(CXX_BIN_DIR ${CMAKE_CXX_COMPILER} DIRECTORY)
  get_filename_component(CXX_INSTALL_DIR ${CXX_BIN_DIR} DIRECTORY)
  set(MFRONT_LD_LIBRARY_PATH "${MFRONT_INSTALL_DIR}/lib")
  if (EXISTS "${CXX_INSTALL_DIR}/lib64")
    string(APPEND MFRONT_LD_LIBRARY_PATH ":${CXX_INSTALL_DIR}/lib64")
  endif ()

  set(VONMISES_MFRONT_FILE ${CMAKE_CURRENT_SOURCE_DIR}/mfront/VonMises.mfront)
  set(VONMISES_MFRONT_DIR ${CMAKE_CURRENT_BINARY_DIR}/mfront)
  set(VONMISES_GENERATED_SOURCES
    ${VONMISES_MFRONT_DIR}/src/VonMises.cxx
    ${VONMISES_MFRONT_DIR}/src/VonMises-generic.cxx)

  add_custom_command(
    OUTPUT ${VONMISES_GENERATED_SOURCES}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${VONMISES_MFRONT_DIR}
    COMMAND ${CMAKE_COMMAND} -E env "LD_LIBRARY_PATH=${MFRONT_LD_LIBRARY_PATH}"
            ${MFRONT_EXECUTABLE} --interface=generic ${VONMISES_MFRONT_FILE}
    DEPENDS ${VONMISES_MFRONT_FILE}
    WORKING_DIRECTORY ${VONMISES_MFRONT_DIR}
    COMMENT "Generating the VonMises MFront behaviour")

  add_library(ArcaneFemVonMises SHARED ${VONMISES_GENERATED_SOURCES})
  # The link against MGIS provides the TFEL include paths required to
  # compile the sources generated by mfront.
  target_link_libraries(ArcaneFemVonMises PRIVATE mgis::MFrontGenericInterface)
  target_include_directories(ArcaneFemVonMises PRIVATE
    $<TARGET_PROPERTY:mgis::MFrontGenericInterface,INTERFACE_INCLUDE_DIRECTORIES>
    ${VONMISES_MFRONT_DIR}/include)
  set_target_properties(ArcaneFemVonMises PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
    OUTPUT_NAME VonMises
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF)
  add_dependencies(Elastoplasticity2 ArcaneFemVonMises)
else ()
  if (NOT MFrontGenericInterface_FOUND AND NOT MFRONT_EXECUTABLE)
    message(STATUS "Elastoplasticity2: MGIS or mfront not found, "
           "the VonMisesMGIS constitutive law is disabled")
  endif ()
endif ()
